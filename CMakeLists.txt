# Preserve bundled library notices, including stb and bimg's optional codecs.
foreach(GM_DEP bgfx bx bimg)
  file(GLOB_RECURSE GM_NOTICES RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}/third_party/${GM_DEP}"
    "third_party/${GM_DEP}/LICENSE*" "third_party/${GM_DEP}/COPYING*" "third_party/${GM_DEP}/COPYRIGHT*")
  foreach(GM_NOTICE IN LISTS GM_NOTICES)
    get_filename_component(GM_NOTICE_DIR "${GM_NOTICE}" DIRECTORY)
    install(FILES "third_party/${GM_DEP}/${GM_NOTICE}" DESTINATION "licenses/${GM_DEP}/${GM_NOTICE_DIR}")
  endforeach()
endforeach()
